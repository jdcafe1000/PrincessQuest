(function(){

  var DAYS=["Mon","Tue","Wed","Thu","Fri","Sat","Sun"];

      q("🍽️","Help set the table"),

      q("🦄","Brave challenge: try something new")

    ]};

  }

  var $=function(id){return document.getElementById(id)};

    var n=document.createElement(tag);

    for(var k in attrs||{}){

      if(k==="text") n.textContent=attrs[k];

      else if(k==="html") n.innerHTML=attrs[k];

      else if(k.slice(0,2)==="on") n.addEventListener(k.slice(2),attrs[k]);

      else if(attrs[k]!==false&&attrs[k]!=null) n.setAttribute(k,attrs[k]);

    }

    (kids||[]).forEach(function(c){n.appendChild(c)});

    return n;

  }

    try{localStorage.setItem(KEY,JSON.stringify(state))}catch(e){}

    if(!db) return;

    if(saving){dirty=true;return}

    saving=true;

      do{

        dirty=false;

      }while(dirty);

      saving=false;

  }

            oninput:function(e){x.value=e.target.value.replace(/[^0-9.]/g,""); renderProgress(); save()}})]));

        extra.appendChild(el("button",{class:"remove",type:"button",text:"✕","aria-label":"Remove "+x.text,

          onclick:function(){state.quests.splice(qi,1); renderQuests(); renderProgress(); save()}}));

      }else if(parseFloat(x.value)>0){

      }

      li.appendChild(extra);

      var days=el("div",{class:"days",role:"group","aria-label":x.text});

      DAYS.forEach(function(d,di){

        days.appendChild(el("button",{class:"day",type:"button","aria-pressed":String(!!x.days[di]),"aria-label":d,html:d+STAR,

            x.days[di]=!x.days[di];

            var b=e.currentTarget; b.setAttribute("aria-pressed",String(x.days[di]));

            b.classList.remove("pop"); if(x.days[di]){void b.offsetWidth; b.classList.add("pop")}

            li.classList.toggle("done",x.days.every(Boolean));

            renderProgress(); save();

          }}));

      });

      li.appendChild(days);

      li.appendChild(el("div",{class:"ribbon",text:"A perfect week for this quest!"}));

      list.appendChild(li);

    });

    if(editing) list.appendChild(el("li",{style:"display:grid"},[el("button",{class:"btn add",type:"button",text:"Add a quest",

      onclick:function(){state.quests.push(q("🌟","")); renderQuests(); renderProgress(); save();

        var ins=list.querySelectorAll(".text input"); if(ins.length) ins[ins.length-1].focus()}})]));

  }


  function render(){

    renderHeader(); renderQuests(); renderProgress();

    $("chest").value=state.chest||"";

    $("edit").textContent=editing?"Done editing":"Edit quests";

  }


  $("chest").addEventListener("input",function(e){state.chest=e.target.value.replace(/[^0-9.]/g,""); e.target.value=state.chest; save()});

    editing=!editing; render();

    $("note").textContent=editing?"Rename quests, tap a picture to change it, and set dollars per star if you like.":"";

  });

    if(!armed){armed=true; e.currentTarget.textContent="Tap again to clear all stars";
